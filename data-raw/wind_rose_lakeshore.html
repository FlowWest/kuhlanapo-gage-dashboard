Import Data

Pull the point time series for Wind_E and Wind_N (eastward and northward 10-m wind vector components, m/s) from NLDAS_FORA0125_H.

This uses the same interface as scripts/fetch_nldas_giovanni.R (the app’s production precip pipeline): NASA GES DISC’s Earthdata Cloud OPeNDAP service (Hyrax), authenticated via Earthdata Login netrc + session cookie, addressed by CMR collection concept-id with a deterministic per-hour granule id (no CMR lookup needed per request). GES DISC retired the old on-premises OPeNDAP service (hydro1.gesdisc.eosdis.nasa.gov) in July 2026 - it now 410s - so this is the current working access path, not the older Giovanni proxy-timeseries endpoint (long dead; never worked reliably) or the retired host.

Data are cached locally as an Rds so the historical pull only needs to run once; set PULL_FRESH_DATA <- TRUE below to force a full re-pull (e.g. after changing the date range or cadence below).

rds_path <- here::here("data-raw", "wind_ts_lakeshore.rds")

PULL_FRESH_DATA <- FALSE

# Historical period + sampling cadence used to build the climatological wind
# rose. Fixed to Water Years 2007-2025 (Oct 1 2006 - Sep 30 2025, local time)
# to match the span of the lake gage record. NLDAS_FORA0125_H is hourly, but
# a full hourly pull over 19 years is very slow (~0.3 s/request against this
# service - hourly would be ~14 hrs). Sampling every 3 hours (8x/day) covers
# the full seasonal and diurnal cycle at roughly a third of that cost
# (~4.5 hrs).
HIST_START_TIME <- with_tz(ymd_hms("2006-10-01 00:00:00", tz = "America/Los_Angeles"), "UTC")
HIST_END_TIME   <- with_tz(ymd_hms("2025-09-30 23:00:00", tz = "America/Los_Angeles"), "UTC")
SAMPLE_CADENCE_HOURS <- 3

# NLDAS_FORA0125_H.2.0 collection concept-id (verified via CMR granule search)
COLLECTION_ID <- "C2033151148-GES_DISC"
BASE_URL <- "https://opendap.earthdata.nasa.gov/collections"

# NLDAS_FORA0125 native grid (see batch_download_nldas.R in interoperable-flows)
GRID_LONS <- seq(-124.9375, -67.0625, 0.125)
GRID_LATS <- seq(25.0625, 52.9375, 0.125)

NETRC_PATH  <- path.expand("~/.netrc")
COOKIE_PATH <- tempfile(fileext = ".cookies")  # shared session cookie, established once

# NASA Earthdata caps concurrent connections per account at 5
MAX_WORKERS <- 5
earthdata_user <- Sys.getenv("EARTHDATA_USER")
earthdata_password <- Sys.getenv("EARTHDATA_PASSWORD")

if (nzchar(earthdata_user) && nzchar(earthdata_password)) {
  writeLines(
    c(
      "machine urs.earthdata.nasa.gov",
      paste("login", earthdata_user),
      paste("password", earthdata_password)
    ),
    NETRC_PATH
  )
} else if (!file.exists(NETRC_PATH)) {
  stop("No EARTHDATA_USER/EARTHDATA_PASSWORD in environment, and no ~/.netrc found")
}

nearest_idx <- function(lon, lat) {
  list(
    x = which.min(abs(GRID_LONS - lon)) - 1L,  # OPeNDAP indices are 0-based
    y = which.min(abs(GRID_LATS - lat)) - 1L
  )
}

build_nldas_url <- function(dt, x_idx, y_idx, vars) {
  yyyymmdd <- format(dt, "%Y%m%d")
  hhhh     <- sprintf("%02d00", as.integer(format(dt, "%H")))
  granule_id <- sprintf("NLDAS_FORA0125_H.2.0:NLDAS_FORA0125_H.A%s.%s.020.nc", yyyymmdd, hhhh)
  query <- paste(
    sprintf("%s%%5B0:1:0%%5D%%5B%d%%5D%%5B%d%%5D", vars, y_idx, x_idx),
    collapse = ","
  )
  sprintf(
    "%s/%s/granules/%s.ascii?%s",
    BASE_URL, COLLECTION_ID, utils::URLencode(granule_id, reserved = TRUE), query
  )
}

parse_opendap_ascii <- function(txt, vars) {
  lines <- str_split(str_trim(txt), "\n")[[1]]
  out <- setNames(rep(NA_real_, length(vars)), vars)
  for (v in vars) {
    pat <- paste0("^", v, "[.]", v, "\\[.*\\],\\s*(-?[0-9.eE+-]+)\\s*$")
    m <- str_subset(lines, pat)
    if (length(m) > 0) out[v] <- as.numeric(str_match(m[1], pat)[, 2])
  }
  out
}

# Do the netrc+redirect handshake once up front and keep the resulting session
# cookie, rather than repeating the full URS login/redirect on every request -
# subsequent requests just present this cookie directly.
establish_session <- function() {
  probe_url <- build_nldas_url(HIST_START_TIME, 0, 0, "Rainf")
  res <- GET(
    probe_url,
    config(netrc = TRUE, netrc_file = NETRC_PATH, followlocation = TRUE,
           cookiefile = COOKIE_PATH, cookiejar = COOKIE_PATH),
    timeout(60)
  )
  if (status_code(res) != 200) {
    stop("Failed to establish EDL session (HTTP ", status_code(res), ") - check EARTHDATA_USER/EARTHDATA_PASSWORD")
  }
  invisible(NULL)
}

fetch_one_hour <- function(dt, x_idx, y_idx, vars) {
  url <- build_nldas_url(dt, x_idx, y_idx, vars)
  # read the already-established shared session cookie, but write any
  # cookie updates to a throwaway per-call file so parallel workers never
  # write-race the shared one
  scratch_jar <- tempfile(fileext = ".cookies")
  on.exit(unlink(scratch_jar), add = TRUE)
  res <- GET(
    url,
    config(netrc = TRUE, netrc_file = NETRC_PATH, followlocation = TRUE,
           cookiefile = COOKIE_PATH, cookiejar = scratch_jar),
    timeout(60)
  )
  if (status_code(res) != 200) {
    return(setNames(rep(NA_real_, length(vars)), vars))
  }
  parse_opendap_ascii(content(res, "text", encoding = "UTF-8"), vars)
}

safe_fetch_one_hour <- possibly(fetch_one_hour, otherwise = NULL, quiet = TRUE)
if (PULL_FRESH_DATA || !file.exists(rds_path)) {

  idx <- nearest_idx(wind_site$lon, wind_site$lat)
  hours_seq <- seq(HIST_START_TIME, HIST_END_TIME, by = paste(SAMPLE_CADENCE_HOURS, "hours"))
  message("Fetching ", length(hours_seq), " timestamps at ", SAMPLE_CADENCE_HOURS,
          "-hour cadence from ", HIST_START_TIME, " to ", HIST_END_TIME)

  establish_session()
  plan(multisession, workers = min(MAX_WORKERS, future::availableCores()))

  results <- future_map(
    hours_seq,
    ~ safe_fetch_one_hour(.x, idx$x, idx$y, c("Wind_E", "Wind_N")),
    .options = furrr_options(seed = TRUE)
  )

  plan(sequential)

  wind_ts <- tibble(site = wind_site$id, timestamp_utc = hours_seq) |>
    bind_cols(bind_rows(results)) |>
    rename(wind_e_ms = Wind_E, wind_n_ms = Wind_N) |>
    filter(!is.na(wind_e_ms), !is.na(wind_n_ms)) |>
    mutate(timestamp = with_tz(timestamp_utc, "America/Los_Angeles")) |>
    arrange(site, timestamp_utc)

  wind_ts |> saveRDS(rds_path)
  message("Saved ", nrow(wind_ts), " total wind records")

} else {

  wind_ts <- readRDS(rds_path)

}

Compute Wind Direction

Convert the eastward/northward wind vector components (m/s) to wind speed (converted to mph) and meteorological wind direction (degrees clockwise from north, direction the wind is blowing from).

MS_TO_MPH <- 2.236936

wind_ts <- wind_ts |>
  mutate(
    wind_speed_mph = sqrt(wind_e_ms^2 + wind_n_ms^2) * MS_TO_MPH,
    wind_dir_deg = (atan2(wind_e_ms, wind_n_ms) * 180 / pi + 180) %% 360
  )

# Custom meteorological-season labels (Mar-May / Jun-Aug / Sep-Nov / Dec-Feb)
# in place of openair's default "spring (MAM)" etc. Named "period" rather
# than "season" since openair reserves that column name for its own
# built-in MAM/JJA/SON/DJF cutData logic.
wind_df <- wind_ts |>
  mutate(
    period = case_when(
      month(timestamp) %in% 3:5  ~ "Mar-May",
      month(timestamp) %in% 6:8  ~ "Jun-Aug",
      month(timestamp) %in% 9:11 ~ "Sep-Nov",
      TRUE                       ~ "Dec-Feb"
    ),
    period = factor(period, levels = c("Mar-May", "Jun-Aug", "Sep-Nov", "Dec-Feb"))
  ) |>
  transmute(date = timestamp, ws = wind_speed_mph, wd = wind_dir_deg, period) |>
  filter(!is.na(ws), !is.na(wd))
